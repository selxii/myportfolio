import pandas as pd
import gspread

# MANUKAN NG BAYAN Sales Automation
def process_sales_data(sheet_url):
    gc = gspread.service_account()
    sh = gc.open_by_url(sheet_url)
    df = pd.DataFrame(sh.sheet1.get_all_records())
    
    sql_statements = []
    for _, row in df.iterrows():
        query = f"INSERT INTO sales VALUES ('{row['date']}', {row['total']});"
        sql_statements.append(query)
        
    return f"Processed {len(sql_statements)} rows cleanly!"
